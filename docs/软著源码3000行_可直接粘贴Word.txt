<script setup>
import { computed, ref, watch } from 'vue'
import { ALL_COURSE_TITLE } from '../../utils/courseMeta'
const props = defineProps({
  modelValue: {
    type: String,
    default: '',
  },
  suggestions: {
    type: Array,
    default: () => [],
  },
  courses: {
    type: Array,
    default: () => [ALL_COURSE_TITLE],
  },
  courseFilter: {
    type: String,
    default: ALL_COURSE_TITLE,
  },
})
const emit = defineEmits(['update:modelValue', 'select', 'update:courseFilter'])
const inputRef = ref(null)
const focused = ref(false)
let blurTimer = null
const showSuggestions = computed(() => focused.value && props.modelValue && props.suggestions.length)
function focusInput() {
  inputRef.value?.focus()
}
function handleKeydown(event) {
  if (event.key === 'Enter' && props.suggestions[0]) {
    emit('select', props.suggestions[0])
    focused.value = false
  }
}
function handleBlur() {
  blurTimer = window.setTimeout(() => {
    focused.value = false
  }, 120)
}
function handleSuggestionMouseDown(item) {
  if (blurTimer) {
    window.clearTimeout(blurTimer)
  }
  emit('select', item)
  focused.value = false
}
watch(
  () => props.modelValue,
  (value) => {
    if (!value) focused.value = false
  },
)
defineExpose({ focusInput })
</script>

function upsertKnowledgeLookup(lookup, node) {
  const routeKnowledgeId = String(node?.routeKnowledgeId || '').trim()
  if (!routeKnowledgeId) return
  const label = String(node?.canonicalName || node?.name || routeKnowledgeId).trim() || routeKnowledgeId
  lookup[routeKnowledgeId] = {

async function refreshRecommendationContext(routeKnowledgeId) {
  const requestId = ++recommendationRequestId
  if (routeKnowledgeId) {
    await graphStore.ensureNodeDetail(routeKnowledgeId)
    const detail = graphStore.detailCache[routeKnowledgeId]
    const target = getRecommendationTarget(detail)
    const recommendation = await graphStore.ensureRecommendedPath(target, routeKnowledgeId, 'detail')
    if (requestId !== recommendationRequestId) return
    applyRecommendationState(recommendation)
    return
  }
  const recommendation = await graphStore.ensureRecommendedPath('', '', 'assistant')
  if (requestId !== recommendationRequestId) return
  applyRecommendationState(recommendation)
}

function getRevealOptions() {
  return {
    anchorPosition: graphCanvasRef.value?.getViewportCenterWorld?.() || null,

function queueExpandedClusterFocus(nodeId, childIds = [], options = {}) {
  graphCanvasRef.value?.queueViewportFocus?.(nodeId, {
    zoomLevel: options.zoomLevel ?? 1.06,
    duration: options.duration ?? 220,
    lockDuration: options.lockDuration ?? 880,
    relatedNodeIds: childIds,
    mode: childIds.length ? 'keep-visible' : 'center',
  })
}
async function revealAndFocus(routeKnowledgeId, options = {}) {
  const node = await graphStore.revealNodeByRouteKnowledgeId(routeKnowledgeId, getRevealOptions())
  if (node?.id) {
    await graphCanvasRef.value?.focusElement?.(node.id, options.duration ?? 200)
    if (options.flash) {
      await graphCanvasRef.value?.flashNode?.(node.id)
    }
  }
  return node
}
function resolveAssistantTarget(target) {
  const normalized = String(target || '').trim()
  if (!normalized) return null
  const direct = graphStore.routeNodeMap.get(normalized)
  if (direct?.routeKnowledgeId) return direct
  const candidate = graphStore.searchNodes(normalized)[0]
  if (candidate?.routeKnowledgeId) {
    return graphStore.routeNodeMap.get(candidate.routeKnowledgeId) || candidate
  }
  return null
}
async function expandKnowledgeFromAssistant(target) {
  const resolved = resolveAssistantTarget(target)
  if (!resolved?.routeKnowledgeId) return
  const node = await revealAndFocus(resolved.routeKnowledgeId, { flash: true, duration: 220 })
  if (!node?.id) return
  if (!graphStore.isNodeExpanded(node.id)) {
    const anchorSnapshot = graphCanvasRef.value?.getNodeSnapshot?.(node.id)
    const occupiedNodes = graphCanvasRef.value?.getNodeSnapshots?.() || []
    const newChildIds = await graphStore.expandNode(node.id, {
      anchorPosition: anchorSnapshot,
      viewportCenterWorld: graphCanvasRef.value?.getViewportCenterWorld?.() || null,
      occupiedNodes,
    })
    queueExpandedClusterFocus(node.id, newChildIds, {
      zoomLevel: 1.08,
      duration: 220,
      lockDuration: 820,
    })
  }
  await graphStore.selectNode(node.id)
  rightCollapsed.value = false
}
async function handleAssistantActions(actions = []) {
  for (const action of actions) {
    const type = String(action?.type || '').trim()
    if (!type) continue
    if (type === 'focus_knowledge') {
      const resolved = resolveAssistantTarget(action.target)
      if (resolved?.routeKnowledgeId) {
        await revealAndFocus(resolved.routeKnowledgeId, {
          flash: action.flash !== false,
          duration: 220,
        })
        rightCollapsed.value = false
      }
      continue
    }
    if (type === 'expand_knowledge') {
      await expandKnowledgeFromAssistant(action.target)
      continue
    }
    if (type === 'highlight_prerequisite_chain') {
      const resolved = resolveAssistantTarget(action.target)
      if (resolved?.routeKnowledgeId) {
        await revealAndFocus(resolved.routeKnowledgeId, { flash: true, duration: 200 })
        await graphStore.ensureNodeDetail(resolved.routeKnowledgeId)
        const path = graphStore.getPrerequisiteRouteChain(resolved.routeKnowledgeId)
        graphStore.highlightPathNodes(path)
        rightCollapsed.value = false
      }
      continue
    }
    if (type === 'open_knowledge_detail') {
      const resolved = resolveAssistantTarget(action.target)
      if (resolved?.routeKnowledgeId) {
        router.push(`/knowledge/${encodeURIComponent(resolved.routeKnowledgeId)}`)
      }
      continue
    }
    if (type === 'filter_course') {
      const courseTitle = String(action.courseTitle || action.target || '').trim()
      if (!courseTitle) continue
      graphStore.setCourseFilter(courseTitle)
      const node = await graphStore.focusCourseRepresentative(courseTitle, getRevealOptions())
      if (node?.id) {
        await graphStore.selectNode(node.id)
        await graphCanvasRef.value?.focusElement?.(node.id, 180)
      }
      continue
    }
    if (type === 'fit_graph_view') {
      await graphCanvasRef.value?.fitView?.()
      continue
    }
    if (type === 'reset_graph_scene') {
      await handleSceneReset()
    }
  }
}

async function handleContextAction({ action, node }) {
  const routeKnowledgeId = node?.data?.routeKnowledgeId
  if (!routeKnowledgeId) return
  if (action === 'toggle-bookmark') {
    userStore.toggleBookmark(routeKnowledgeId)
    return
  }
  if (action === 'note') {
    await revealAndFocus(routeKnowledgeId)
    rightCollapsed.value = false
    return
  }
  if (action === 'detail') {
    router.push(`/knowledge/${encodeURIComponent(routeKnowledgeId)}`)
    return
  }
  if (action === 'ai') {
    await revealAndFocus(routeKnowledgeId)
    rightCollapsed.value = false
    aiAutoPrompt.value = {
      id: Date.now(),
      text: `请帮我解释知识点“${routeKnowledgeId}”，并结合前置知识给出学习建议。`,
    }
    aiOpen.value = true
    return
  }
  if (action === 'chain') {
    await graphStore.ensureNodeDetail(routeKnowledgeId)
    const path = graphStore.getPrerequisiteRouteChain(routeKnowledgeId)
    graphStore.highlightPathNodes(path)
    return
  }
  if (action === 'mastered') {
    userStore.markMastered(routeKnowledgeId)
    return
  }
  if (action === 'add-to-path') {
    const path = userStore.customPaths[0] || userStore.createCustomPath('我的学习路径')
    userStore.addToPath(path.id, routeKnowledgeId)
  }
}

  if (!path) return
  if (toIndex < 0 || toIndex >= path.nodes.length || fromIndex === toIndex) return
  userStore.reorderPath(pathId, fromIndex, toIndex)
}

function getStatusMeta(routeKnowledgeId) {
  const progress = userStore.getProgressMeta(routeKnowledgeId)
  const mastery = Number(progress.masteryScore || 0)
  const answeredCount = Number(progress.answeredCount || 0)
  if (mastery >= 80) return '✅'
  if (mastery < 30 && answeredCount > 0) return '⚠️'
  if (mastery >= 40) return '📖'
  return '⬜'
}

  const currentCourse = normalizeCourseTitle(props.detail?.chapterTitle, DEFAULT_COURSE_TITLE)
  return (props.recommendedPath || [])
    .map((item) => {
      const node = graphStore.routeNodeMap.get(item.knowledgeId || item.title)
      const displayTitle =
        String(node?.canonicalName || node?.name || item.title || item.knowledgeId || '').trim()
      return {
        ...item,
        displayTitle,
        sourceCourse: normalizeCourseTitle(item.sourceCourse || currentCourse, DEFAULT_COURSE_TITLE),
        targetCourse: normalizeCourseTitle(item.targetCourse || node?.chapterTitle, DEFAULT_COURSE_TITLE),
        courseLabel:
          item.reasonCode === 'cross_course_transfer'
            ? `${normalizeCourseTitle(item.sourceCourse || currentCourse, DEFAULT_COURSE_TITLE)} → ${normalizeCourseTitle(item.targetCourse || node?.chapterTitle, DEFAULT_COURSE_TITLE)}`
            : normalizeCourseTitle(item.targetCourse || node?.chapterTitle, DEFAULT_COURSE_TITLE),
      }
    })
    .filter((item) => {
      if (!item?.title && !item?.knowledgeId) return false
      if (item.reasonCode === 'cross_course_transfer' && item.sourceCourse && item.targetCourse) {
        return item.sourceCourse !== item.targetCourse
      }
      return Boolean(item.targetCourse && item.targetCourse !== currentCourse)
    })
    .slice(0, 3)
})

function clamp(value, min, max) {
  return Math.max(min, Math.min(max, value))
}
function hexToRgb(color = '#CBD5E1') {
  const normalized = String(color || '').trim()
  if (!normalized.startsWith('#')) return null
  let hex = normalized.slice(1)
  if (hex.length === 3) hex = hex.split('').map((char) => char + char).join('')
  if (hex.length !== 6) return null
  const red = Number.parseInt(hex.slice(0, 2), 16)
  const green = Number.parseInt(hex.slice(2, 4), 16)
  const blue = Number.parseInt(hex.slice(4, 6), 16)
  return { red, green, blue }
}
function mixCourseColor(fromColor, toColor, ratio, alpha = 1) {
  const left = hexToRgb(fromColor)
  const right = hexToRgb(toColor)
  const t = clamp(Number(ratio || 0), 0, 1)
  if (!left || !right) return String(toColor || fromColor || '#CBD5E1')
  const red = Math.round(left.red + (right.red - left.red) * t)
  const green = Math.round(left.green + (right.green - left.green) * t)
  const blue = Math.round(left.blue + (right.blue - left.blue) * t)
  return `rgba(${red}, ${green}, ${blue}, ${clamp(alpha, 0, 1)})`
}

function splitLabelLines(label = '', lineLength = 5, maxLines = 3) {
  const cleaned = String(label || '').replace(/\s+/g, ' ').trim()
  if (!cleaned) return ['未命名']
  const lines = []
  for (let index = 0; index < cleaned.length; index += lineLength) {
    if (lines.length >= maxLines) break
    lines.push(cleaned.slice(index, index + lineLength))
  }
  if (cleaned.length > lineLength * maxLines && lines.length) {
    const last = lines[lines.length - 1]
    lines[lines.length - 1] = `${last.slice(0, Math.max(1, last.length - 1))}…`
  }
  return lines
}
function estimateCollisionRadius(label = '', radius = 32, options = {}) {
  const isRoot = Boolean(options.isRoot)
  const isExpanded = Boolean(options.isExpanded)
  const fontSize = isRoot ? 13.8 : 10.9
  const lines = splitLabelLines(label, isRoot ? 6 : 5, 3)
  const widest = Math.max(...lines.map((line) => String(line || '').length), 1)
  const textWidth = widest * fontSize * 1.04
  const textHeight = lines.length * (fontSize + 5)
  const labelRadius = Math.hypot(textWidth * 0.58, textHeight * 0.68)
  const labelSpreadBoost = Math.min(
    isRoot ? 32 : 24,
    Math.max(0, widest - 3) * (isRoot ? 2.8 : 2.3) + Math.max(0, lines.length - 1) * 5,
  )
  const basePadding = isRoot ? (isExpanded ? 32 : 28) : isExpanded ? 24 : 21
  return Math.ceil(
    Math.max(radius + basePadding, labelRadius + radius * (isRoot ? 0.46 : 0.4) + labelSpreadBoost),
  )
}

}

export function transformGraphData(nodes = [], edges = [], options = {}) {
  const expandedSet = new Set(options.expandedNodeIds || [])
  const pinnedSet = new Set(options.pinnedNodeIds || [])
  const nodeMap = new Map(nodes.map((node) => [node.id, node]))
  const visibleNodes = [...nodes]
    .sort((left, right) => {
      const rootDelta = Number(right.isRoot) - Number(left.isRoot)
      if (rootDelta !== 0) return rootDelta
      const depthDelta = Number(left.depth || 0) - Number(right.depth || 0)
      if (depthDelta !== 0) return depthDelta
      return getNodeDisplayName(left).localeCompare(getNodeDisplayName(right), 'zh-CN')
    })
    .map((node) =>
      normalizeGraphNode(node, {
        progressMap: options.progressMap || {},
        expandedSet,
        pinnedSet,
        selectedNodeId: options.selectedNodeId || null,
      }),
    )
  const visibleEdges = edges
    .filter((edge) => nodeMap.has(edge.sourceId) && nodeMap.has(edge.targetId))
    .map((edge) => normalizeGraphEdge(edge, nodeMap))
  return {
    nodes: visibleNodes,
    edges: visibleEdges,
    meta: {
      totalNodes: Number(options.totalNodes || visibleNodes.length),
      visibleNodes: visibleNodes.length,
      visibleEdges: visibleEdges.length,
      rootCount: visibleNodes.filter((node) => node.data?.isRoot).length,
      expandedCount: expandedSet.size,
      rootNodeIds: [...(options.rootNodeIds || [])],
    },
  }
}

function normalizeKeyword(value = '') {
  return String(value || '').trim().toLowerCase().replace(/\s+/g, '')
}
function scoreResult(item, query) {
  const name = normalizeKeyword(item.name)
  const canonical = normalizeKeyword(item.canonicalName)
  const chapter = normalizeKeyword(item.chapterTitle)
  if (!query) return 0
  if (canonical === query || name === query) return 100
  if (canonical.startsWith(query) || name.startsWith(query)) return 80
  if (canonical.includes(query) || name.includes(query)) return 60
  if (chapter.includes(query)) return 30
  return 0
}
export function createSearchIndex(nodes = []) {
  return nodes.map((node) => ({
    id: node.id,
    routeKnowledgeId: node.routeKnowledgeId,
    name: node.name || node.canonicalName || node.routeKnowledgeId || '',
    canonicalName: node.canonicalName || node.name || '',
    chapterTitle: resolveNodeCourseTitle(node, DEFAULT_COURSE_TITLE),
    entityType: node.entityType || 'concept',
  }))
}
export function searchInIndex(index = [], keyword = '', limit = 8) {
  const query = normalizeKeyword(keyword)
  if (!query) return []
  return index
    .map((item) => ({ item, score: scoreResult(item, query) }))
    .filter(({ score }) => score > 0)
    .sort((left, right) => {
      if (right.score !== left.score) return right.score - left.score
      return String(left.item.name).localeCompare(String(right.item.name), 'zh-CN')
    })
    .slice(0, limit)
    .map(({ item }) => item)
}
function buildLookupVariants(value = '') {
  const normalized = String(value || '').trim()
  if (!normalized) return []
  const lower = normalized.toLowerCase()
  return lower === normalized ? [normalized] : [normalized, lower]
}
function readLookupLabel(entry) {
  if (!entry) return ''
  if (typeof entry === 'string') {
    return String(entry).trim()
  }
  return String(entry.label || entry.canonicalName || entry.name || '').trim()
}
export function looksLikeMachineKnowledgeId(value = '') {
  const normalized = String(value || '').trim()
  return /^[0-9a-f]{16,}$/i.test(normalized)
}
export function extractKnowledgeIdFromRoute(route = '') {
  const normalized = String(route || '').trim()
  const match = normalized.match(/\/knowledge\/([^/?#]+)/)
  return match ? decodeURIComponent(match[1]) : ''
}
export function buildKnowledgeTitleLookup(nodes = []) {
  const lookup = {}
  ;(nodes || []).forEach((node) => {
    const label =
      String(node?.canonicalName || node?.name || node?.routeKnowledgeId || node?.id || '').trim()
    if (!label) return
    const keys = [
      node?.routeKnowledgeId,
      node?.id,
      node?.canonicalName,
      node?.name,
      extractKnowledgeIdFromRoute(node?.route),
    ]
    keys.forEach((key) => {
      buildLookupVariants(key).forEach((variant) => {
        const existing = readLookupLabel(lookup[variant])
        if (!existing || looksLikeMachineKnowledgeId(existing)) {
          lookup[variant] = { label }
        }
      })
    })
  })
  return lookup
}
export function resolveKnowledgeTitle(value = '', lookup = {}) {
  for (const variant of buildLookupVariants(value)) {
    const label = readLookupLabel(lookup?.[variant])
    if (label) return label
  }
  return ''
}
export function replaceKnowledgeIdsInText(text = '', lookup = {}) {
  return String(text || '').replace(/\b[0-9a-f]{16,}\b/gi, (matched) => {
    const label = resolveKnowledgeTitle(matched, lookup)
    return label || matched
  })
}
export function resolveRecommendationTitle(item = {}, lookup = {}) {
  const rawTitle = String(item?.title || '').trim()
  const candidates = [
    item?.knowledgeId,
    item?.routeKnowledgeId,
    extractKnowledgeIdFromRoute(item?.route),
    rawTitle,
  ]
  const resolvedLabel = candidates.map((value) => resolveKnowledgeTitle(value, lookup)).find(Boolean) || ''
  if (resolvedLabel && (!rawTitle || looksLikeMachineKnowledgeId(rawTitle))) {
    return resolvedLabel
  }
  const normalizedTitle = replaceKnowledgeIdsInText(rawTitle, lookup)
  return normalizedTitle || resolvedLabel || String(item?.knowledgeId || '').trim() || '推荐知识点'
}
export function normalizeRecommendationPresentation(item = {}, lookup = {}) {
  const displayTitle = resolveRecommendationTitle(item, lookup)
  const displayDetail = replaceKnowledgeIdsInText(item?.detail || '', lookup)
  const sourceCourse = String(item?.sourceCourse || '').trim()
  const targetCourse = String(item?.targetCourse || '').trim()
  return {
    ...item,
    title: displayTitle,
    detail: displayDetail,
    displayTitle,
    displayDetail,
    displayReasonLabel: String(item?.reasonLabel || '').trim() || '图谱推荐',
    displayCourseLabel:
      sourceCourse && targetCourse && sourceCourse !== targetCourse
        ? `${sourceCourse} → ${targetCourse}`
        : targetCourse || sourceCourse || '',
  }
}

function buildKnowledgeCatalog() {
  const byRouteId = {}
  const aliasToRouteId = {}
  for (const domain of domains) {
    for (const point of domain.points) {
      const entry = {
        id: point.id,
        route: point.route,
        name: point.name,
        domainId: domain.id,
        domainName: domain.name,
        level: point.level,
        levelLabel: point.levelLabel,
        description: point.description,
        mastery: point.mastery,
      }
      byRouteId[point.id] = entry
      const aliases = new Set([
        point.id,
        point.name,
        point.route?.replace('/knowledge/', ''),
        ...(MANUAL_ALIASES[point.id] || []),
      ])
      for (const alias of aliases) {
        const key = normalizeLookupKey(alias)
        if (!key) continue
        aliasToRouteId[key] = point.id
      }
    }
  }
  return { byRouteId, aliasToRouteId }
}

    null
  )
}

function stringHash(value = '') {
  return [...String(value)].reduce((hash, char) => hash + char.charCodeAt(0), 0)
}
function withAlpha(hex = '#64748B', alpha = '22') {
  const normalized = String(hex || '').trim()
  return normalized.startsWith('#') && normalized.length === 7 ? `${normalized}${alpha}` : normalized
}
export function getCoursePalette(courseTitle = DEFAULT_COURSE_TITLE) {
  const key = normalizeCourseTitle(courseTitle, DEFAULT_COURSE_TITLE)
  if (COURSE_COLOR_MAP[key]) return COURSE_COLOR_MAP[key]
  const seed = stringHash(key)
  const primary = FALLBACK_PALETTE[seed % FALLBACK_PALETTE.length]
  return {
    primary,
    light: withAlpha(primary, '22'),

    dark: primary,
  }
}

export function getCourseOptions(nodes = []) {
  const courses = new Set([ALL_COURSE_TITLE])
  nodes.forEach((node) => {
    courses.add(resolveNodeCourseTitle(node, DEFAULT_COURSE_TITLE))
  })
  const orderIndex = new Map(COURSE_TITLE_ORDER.map((title, index) => [title, index]))
  return [...courses].sort((left, right) => {
    if (left === ALL_COURSE_TITLE) return -1
    if (right === ALL_COURSE_TITLE) return 1
    const leftRank = orderIndex.get(left) ?? Number.MAX_SAFE_INTEGER
    const rightRank = orderIndex.get(right) ?? Number.MAX_SAFE_INTEGER
    if (leftRank !== rightRank) return leftRank - rightRank
    return String(left).localeCompare(String(right), 'zh-CN')
  })
}

import { readStorage, STORAGE_KEYS } from '../utils/storage'
const API_BASE_URL = import.meta.env.VITE_API_BASE_URL || '/api'
const LEGACY_AUTH_STORAGE_KEY = 'course-recommend-auth'
export class ApiError extends Error {
  constructor(message, status = 500, payload = null) {
    super(message)
    this.name = 'ApiError'
    this.status = status
    this.payload = payload
  }
}
function readAuthSession() {
  const primary = readStorage(STORAGE_KEYS.auth, null)
  if (primary && typeof primary === 'object') {
    return primary
  }
  const legacy = readStorage(LEGACY_AUTH_STORAGE_KEY, null)
  if (legacy && typeof legacy === 'object') {
    return legacy
  }
  return null
}
function extractToken(session) {
  if (!session || typeof session !== 'object') return ''
  return (
    session.token ||
    session.accessToken ||
    session?.data?.token ||
    session?.data?.accessToken ||
    session?.user?.token ||
    session?.user?.accessToken ||
    ''
  )
}
function getToken() {
  return String(extractToken(readAuthSession()) || '').trim()
}
function buildHeaders(headers = {}, auth = true) {
  const nextHeaders = { ...headers }
  const token = auth ? getToken() : ''
  if (token) {
    nextHeaders.Authorization = `Bearer ${token}`
  }
  return nextHeaders
}
async function parsePayload(response) {
  const contentType = response.headers.get('content-type') || ''
  if (contentType.includes('application/json')) {
    return response.json()
  }
  return response.text()
}
export async function request(path, options = {}) {
  const {
    auth = true,
    headers = {},
    body,
    ...restOptions
  } = options
  const isFormData = typeof FormData !== 'undefined' && body instanceof FormData
  const requestHeaders = buildHeaders(headers, auth)
  if (body !== undefined && !isFormData && !requestHeaders['Content-Type']) {
    requestHeaders['Content-Type'] = 'application/json'
  }
  const response = await fetch(`${API_BASE_URL}${path}`, {
    ...restOptions,
    headers: requestHeaders,
    body: body === undefined ? undefined : isFormData || typeof body === 'string' ? body : JSON.stringify(body),
  })
  const payload = await parsePayload(response)
  if (!response.ok) {
    const message = payload?.message || payload?.error || '请求失败，请稍后重试。'
    throw new ApiError(message, response.status, payload)
  }
  if (payload && typeof payload === 'object' && 'code' in payload && payload.code !== 0) {
    throw new ApiError(payload.message || '请求失败，请稍后重试。', response.status, payload)
  }
  if (payload && typeof payload === 'object' && 'data' in payload) {
    return payload.data
  }
  return payload
}
export async function streamRequest(path, options = {}) {
  const {
    auth = true,
    headers = {},
    body,
    onMessage,
    onEvent,
    signal,
    ...restOptions
  } = options
  async function openStream(useAuth) {
    return fetch(`${API_BASE_URL}${path}`, {
      ...restOptions,
      headers: {
        Accept: 'text/event-stream',
        ...buildHeaders(headers, useAuth),
        'Content-Type': 'application/json',
      },
      body: body === undefined ? undefined : JSON.stringify(body),
      signal,
    })
  }
  let response = await openStream(auth)
  if (auth && response.status === 401) {
    response = await openStream(false)
  }
  if (!response.ok) {
    const payload = await parsePayload(response).catch(() => null)
    throw new ApiError(payload?.message || payload?.error || '学习助手暂时不可用。', response.status, payload)
  }
  if (!response.body) {
    throw new ApiError('浏览器当前不支持流式响应。', response.status)
  }
  const reader = response.body.getReader()
  const decoder = new TextDecoder('utf-8')
  let buffer = ''
  while (true) {
    const { done, value } = await reader.read()
    if (done) break
    buffer += decoder.decode(value, { stream: true })
    const chunks = buffer.split('\n\n')
    buffer = chunks.pop() || ''
    for (const chunk of chunks) {
      const line = chunk
        .split('\n')
        .map((item) => item.trim())
        .find((item) => item.startsWith('data:'))
      if (!line) continue
      const content = line.replace(/^data:\s*/, '')
      if (content === '[DONE]') return
      const payload = JSON.parse(content)
      const token = payload?.choices?.[0]?.delta?.content || ''
      if (token && typeof onMessage === 'function') {
        onMessage(token, payload)
      }
      if (typeof onEvent === 'function') {
        onEvent(payload)
      }
    }
  }
}

export function fetchFrontierBriefing(query = {}) {
  const normalized =
    typeof query === 'string'
      ? {
          knowledgeId: query,
          knowledge: query,
        }
      : (query || {})
  const params = new URLSearchParams()
  if (normalized.knowledgeId) {
    params.set('knowledgeId', String(normalized.knowledgeId))
  }
  if (normalized.knowledge) {
    params.set('knowledge', String(normalized.knowledge))
  }
  const suffix = params.toString() ? `?${params.toString()}` : ''
  return request(`/frontier/briefing${suffix}`)
}
